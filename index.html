firebase.database().ref("/").on("value", function(snapshot){ var data = snapshot.val(); document.getElementById("status").innerHTML = data.status; if(data.status == "HIGH VIBRATION DETECTED") { document.getElementById("latitude").innerHTML = data.latitude; document.getElementById("longitude").innerHTML = data.longitude; document.getElementById("mapLink").style.display = "inline"; document.getElementById("mapLink").href = "https://maps.google.com/?q=" + data.latitude + "," + data.longitude; } else { document.getElementById("latitude").innerHTML = "Hidden"; document.getElementById("longitude").innerHTML = "Hidden"; document.getElementById("mapLink").style.display = "none"; } });
